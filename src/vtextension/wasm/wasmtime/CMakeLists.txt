# vtextension_wasmtime: the Wasmtime backend of the WebAssembly runtime interface (../WasmRuntime.hpp). The only
# target that links Wasmtime, and built only where CONTOUR_HAVE_WASMTIME is ON. It is not the default backend
# yet: nothing but its own tests links it. scripts/check-wasm-runtime-containment.py keeps it contained.

add_library(vtextension_wasmtime STATIC
    WasmtimeEngine.cpp
    WasmtimeInstance.cpp
    WasmtimeModule.cpp
    WasmtimeEngine.hpp
    WasmtimeHandles.hpp
    WasmtimeMemoryCreator.cpp
    WasmtimeMemoryCreator.hpp
    WasmtimeRuntime.hpp
)
# vtextension_memory_provision for the memory a host provides an instance with (../host/MemoryProvision.hpp).
target_link_libraries(vtextension_wasmtime PUBLIC vtextension PRIVATE vtextension_memory_provision contour::wasmtime)

# The backstop WasmtimeEngine.cpp includes: it stops the build unless the Wasmtime headers have every
# feature the backend switches, from the same list the configure-time probe checks an installed Wasmtime
# against (cmake/ContourThirdParties.cmake).
ContourThirdParties_WasmtimeFeatureChecks(_wasmtime_feature_checks ${_WASMTIME_REQUIRED_FEATURES})
file(CONFIGURE
    OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/generated/vtextension/wasm/wasmtime/WasmtimeRequiredFeatures.hpp"
    CONTENT "// SPDX-License-Identifier: Apache-2.0\n// Generated by CMake from _WASMTIME_REQUIRED_FEATURES (cmake/ContourThirdParties.cmake).\n#pragma once\n\n${_wasmtime_feature_checks}"
    @ONLY
)
target_include_directories(vtextension_wasmtime PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")

# The engine the extension host executable runs (../host/HelperEngine.hpp): Wasmtime's, trapping through guard
# pages and signals -- except where a sanitizer or coverage owns the signal handlers, which trap in software.
add_library(vtextension_wasmtime_helper_engine STATIC
    HelperEngine.cpp
    ${PROJECT_SOURCE_DIR}/src/vtextension/wasm/host/HelperEngine.hpp
)
target_link_libraries(vtextension_wasmtime_helper_engine PUBLIC vtextension_host PRIVATE vtextension_wasmtime)

if(VTEXTENSION_TESTING)
    # Test support for the extension host's suite (vtextension_remote_test): the engine a host process drives,
    # reached through a neutral probe (../testing/HostedEngine.hpp), so that the host's tests name no engine.
    add_library(vtextension_wasmtime_hosted_engine STATIC
        HostedEngine.cpp
        ${PROJECT_SOURCE_DIR}/src/vtextension/wasm/testing/HostedEngine.hpp
    )
    target_link_libraries(vtextension_wasmtime_hosted_engine PUBLIC vtextension_wasmtime)
endif()

if(VTEXTENSION_TESTING AND CONTOUR_WASMTIME_MISSING_TEST_FEATURES)
    list(JOIN CONTOUR_WASMTIME_MISSING_TEST_FEATURES ", " _wasmtime_missing_text)
    message(STATUS "vtextension_wasmtime_test is not built: the installed Wasmtime lacks ${_wasmtime_missing_text}, "
        "which its tests need.")
elseif(VTEXTENSION_TESTING)
    # The backend's own suite: the contract, and what is Wasmtime's alone, which drives its C API directly.
    # PreparedGeneration.cpp defines the probe runtime-agnostic tests read (../testing/PreparedGeneration.hpp).
    add_executable(vtextension_wasmtime_test
        ${PROJECT_SOURCE_DIR}/src/vtextension/test_main.cpp
        PreparedGeneration.cpp
        WasmtimeEngine_test.cpp
    )
    target_link_libraries(vtextension_wasmtime_test PRIVATE
        vtextension vtextension_wasmtime vtextension_memory_provision contour::wasmtime core::testing Catch2::Catch2
        contour_test_trace)
    add_test(NAME vtextension_wasmtime_test COMMAND vtextension_wasmtime_test)
    if(CONTOUR_WASM_GUESTS)
        # The hidden measurement case renders a bundled extension, read from where vtextension_test reads it.
        add_dependencies(vtextension_wasmtime_test contour_bundled_extensions)
        target_compile_definitions(vtextension_wasmtime_test PRIVATE
            VTEXTENSION_TEST_MODULE_DIR="${CONTOUR_CARGO_TARGET_DIR}/${CONTOUR_WASM_TARGET}/release")
    endif()
    if(NOT WIN32)
        # Catch2's TEST_CASE expands __COUNTER__, which Clang flags as a C2y extension under -Werror.
        target_compile_options(vtextension_wasmtime_test PRIVATE -Wno-c2y-extensions)
    endif()
endif()
